    14→function valleySky(ctx, W, H, p, night = 0.62) {
    15→  const g = ctx.createLinearGradient(0, 0, 0, H);
    16→  g.addColorStop(0, "#171b21"); g.addColorStop(0.55, "#20242b"); g.addColorStop(1, "#101216");
    17→  ctx.fillStyle = g; ctx.fillRect(0, 0, W, H);
    18→  // mountain layers (Bergen: Ulriken ridge + valley V)
    19→  const layers = [
    20→    { y: 0.52, c: `rgba(38,44,52,${night})`, f: 2.1, a: 90, s: 1.0 },
    21→    { y: 0.62, c: `rgba(30,35,42,${night + 0.1})`, f: 3.2, a: 60, s: 1.4 },
    22→    { y: 0.74, c: `rgba(22,26,31,${night + 0.2})`, f: 4.5, a: 40, s: 2.0 },
    23→  ];
    24→  layers.forEach((L, li) => {
    25→    ctx.beginPath(); ctx.moveTo(0, H);
    26→    for (let x = 0; x <= W; x += 8) {
    27→      const y = H * L.y - Math.sin(x / W * Math.PI * L.f + li * 2) * L.a * 0.4 - Math.sin(x / 137) * 18;
    28→      ctx.lineTo(x, y);
    29→    }
    30→    ctx.lineTo(W, H); ctx.closePath();
    31→    ctx.fillStyle = L.c; ctx.fill();
    32→  });
    33→  // valley V between two ridges
    34→  ctx.beginPath();
    35→  ctx.moveTo(W * 0.18, H * 0.30); ctx.lineTo(W * 0.46, H * 0.78); ctx.lineTo(W * 0.78, H * 0.28);
    36→  ctx.strokeStyle = "rgba(150,160,170,0.20)"; ctx.lineWidth = 2; ctx.stroke();
    37→  fog(ctx, W, H, p * 30, 5, 0.06);
    38→}

   176→  });

   903→// ===================== master render =====================
   904→export function renderFrame(ctx, W, H, tl, tAbs) {
   905→  const b = tl.blocks.find(bl => tAbs >= bl.start && tAbs < bl.end) || tl.blocks[tl.blocks.length - 1];
   906→  const tLoc = tAbs - b.start;
   907→  chrome(ctx, W, H, tLoc, tAbs, { showCase: b.id !== "b0d" && b.id !== "b8d" });
   908→  const fn = scenes[b.visual] || scenes.closing;
   909→  ctx.save();
   910→  fn(ctx, W, H, b, tLoc, tAbs);
   911→  ctx.restore();
   912→  // screen text layer above art
   913→  ctx.save();
   914→  if (!["title_card", "end_card"].includes(b.visual)) drawScreen(ctx, W, H, b, tLoc / b.dur);
   915→  ctx.restore();
   916→  // edge fade per scene
   917→  ctx.save();
   918→  edgeFade(ctx, W, H, tLoc, b.dur, 0.4, b.visual === "end_card" ? 0 : 0.4);
   919→  ctx.restore();
   920→}
