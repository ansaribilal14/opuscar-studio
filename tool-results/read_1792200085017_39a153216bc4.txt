   240→   240→    }

   242→   242→    const tl = T.landing[0];

   244→   244→    if (C.signature) {
   245→   245→      engraveText(ctx, C.signature.left || '', BORDER[0] + 4, 1026, { size: 13, italic: true, align: 'left', p: seg(t, tl + 1.0, tl + 1.4) });
   246→   246→      engraveText(ctx, C.signature.right || '', BORDER[2] - 4, 1026, { size: 13, italic: true, align: 'right', p: seg(t, tl + 1.2, tl + 1.6) });

   250→   250→  function screenOf(cam, x, y) { const m = new DOMMatrix().translateSelf(W / 2, H / 2).rotateSelf(cam.rot * 180 / Math.PI).scaleSelf(cam.z, cam.z).translateSelf(-cam.cx, -cam.cy); const p = m.transformPoint(new DOMPoint(x, y)); return [p.x, p.y]; }
   251→   251→  function drawCopperScene(ctx, t, cam) {
   252→   252→    applyCam(ctx, cam);
   253→   253→    Cu.drawCopper(ctx, [-300, -300, W + 300, H + 300]);
   254→   254→    Cu.copperLight(ctx, W, H, cam); applyCam(ctx, cam);
   255→   255→    guide.draw(ctx, Infinity, { color: 'rgba(255,226,196,0.4)', wScale: 0.3, dx: -0.25, dy: -0.25 });
   256→   256→    guide.draw(ctx, Infinity, { color: 'rgba(60,24,8,0.25)', wScale: 0.2, dx: 0.2, dy: 0.2 });
   257→   257→    Cu.drawGrooves(ctx, ink, Math.min(t, T.lift + 0.001), { zoom: cam.z });
   258→   258→    const tp = tipAt(Math.min(t, T.lift)), sp = screenOf(cam, tp.x, tp.y), dv = [tp.dx * Math.cos(cam.rot) - tp.dy * Math.sin(cam.rot), tp.dx * Math.sin(cam.rot) + tp.dy * Math.cos(cam.rot)];
   259→   259→    const lift = ss(seg(t, T.lift, T.lift + 0.35));
   260→   260→    const cut = (Math.min(t, T.lift) + 1.6) * 150;
   261→   261→    ctx.save(); ctx.setTransform(1, 0, 0, 1, 0, 0);
   262→   262→    Cu.drawSwarf(ctx, sp, dv, cut, { s: 1.25 });
   263→   263→    Cu.drawBurin(ctx, sp, dv, { s: 1.1, lift });
   264→   264→    ctx.restore();

   267→   267→  function drawSubs(ctx, t) {
   268→   268→    const sb = subs.find(x => t >= x.t0 && t < x.t1); if (!sb) return;
   269→   269→    const a = Math.min(seg(t, sb.t0, sb.t0 + 0.25), 1 - seg(t, sb.t1 - 0.2, sb.t1));
   270→   270→    ctx.save(); ctx.setTransform(1, 0, 0, 1, 0, 0);
   271→   271→    const size = 38, lines = wrapLines(ctx, sb.text, 860, { size, font: FONTS.script });
   272→   272→    const lh = size * 1.12, hgt = lines.length * lh + 24, wid = Math.max(...lines.map(l => measure(ctx, l, { size, font: FONTS.script, weight: 400 }).tw)) + 70;
   273→   273→    const y1 = 1058, y0 = y1 - hgt, x0 = 960 - wid / 2;
   274→   274→    ctx.globalAlpha = a;
   275→   275→    ctx.shadowColor = 'rgba(60,40,20,0.28)'; ctx.shadowBlur = 14; ctx.shadowOffsetY = 4;
   276→   276→    ctx.fillStyle = '#f4eddb'; ctx.beginPath();
   277→   277→    const R = B.RNG(7); ctx.moveTo(x0, y0); for (let x = x0; x <= x0 + wid; x += 14) ctx.lineTo(x, y0 + (R() - 0.5) * 2.2); for (let y = y0; y <= y1; y += 12) ctx.lineTo(x0 + wid + (R() - 0.5) * 2, y); for (let x = x0 + wid; x >= x0; x -= 14) ctx.lineTo(x, y1 + (R() - 0.5) * 2.2); for (let y = y1; y >= y0; y -= 12) ctx.lineTo(x0 + (R() - 0.5) * 2, y); ctx.closePath(); ctx.fill();
   278→   278→    ctx.shadowColor = 'transparent';
   279→   279→    lines.forEach((ln, i) => writeScript(ctx, ln, 960, y0 + 12 + lh * (i + 0.8), { size, color: '#2b1e14', p: 1, alpha: a }));
   280→   280→    ctx.restore();
   281→   281→  }

   297→   297→    ctx.restore();

   300→   300→  function render(ctx, t) {
   301→   301→    ctx.setTransform(1, 0, 0, 1, 0, 0); ctx.globalAlpha = 1; ctx.globalCompositeOperation = 'source-over'; ctx.filter = 'none';
   302→   302→    const cam = camAt(t);
   303→   303→    if (t < T.peel[0]) drawCopperScene(ctx, t, cam);
   304→   304→    else if (t < T.peel[1]) drawPeel(ctx, t, cam);
   305→   305→    else drawPaperScene(ctx, t, cam);
   306→   306→    // print finish: a warm vignette
   307→   307→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   308→   308→    if (t >= T.peel[0]) { const vg = ctx.createRadialGradient(W / 2, H * 0.5, H * 0.55, W / 2, H / 2, H * 1.15); vg.addColorStop(0, 'rgba(255,255,255,0)'); vg.addColorStop(1, 'rgba(120,86,46,0.2)'); ctx.globalCompositeOperation = 'multiply'; ctx.fillStyle = vg; ctx.fillRect(0, 0, W, H); ctx.globalCompositeOperation = 'source-over'; }
   309→   309→    if (t >= T.peel[1] && !(typeof location !== 'undefined' && location.search.includes('nosub'))) drawSubs(ctx, t);
   310→   310→    drawEnd(ctx, t);
   311→   311→  }

   313→   313→  // the proof is pulled: the sheet peels back across the frame. Behind the fold, the printed face; ahead of it, the copper;
   314→   314→  // on the fold, the curling sheet shows its back with the print showing through, mirrored.
   315→   315→  function drawPeel(ctx, t, cam) {
   316→   316→    const u = ss(seg(t, ...T.peel)) * 0.7 + seg(t, ...T.peel) * 0.3, dir = [0.8, 0.6], span = 2700, off = lerp(-span / 2 - 260, span / 2 + 260, u);
   317→   317→    const c = [W / 2 - dir[0] * off, H / 2 - dir[1] * off], n = [-dir[1], dir[0]], fw = 380;
   318→   318→    const band = (d0, d1) => { const A = [c[0] + dir[0] * d0, c[1] + dir[1] * d0], Bp = [c[0] + dir[0] * d1, c[1] + dir[1] * d1]; ctx.beginPath(); ctx.moveTo(A[0] + n[0] * 3000, A[1] + n[1] * 3000); ctx.lineTo(A[0] - n[0] * 3000, A[1] - n[1] * 3000); ctx.lineTo(Bp[0] - n[0] * 3000, Bp[1] - n[1] * 3000); ctx.lineTo(Bp[0] + n[0] * 3000, Bp[1] + n[1] * 3000); ctx.closePath(); };
   319→   319→    drawCopperScene(ctx, t, cam);
   320→   320→    ctx.save(); ctx.setTransform(1, 0, 0, 1, 0, 0); band(0, 4000); ctx.clip(); drawPaperScene(ctx, t, cam, { noColour: true }); ctx.restore();
   321→   321→    // shadow of the lifted sheet on the copper
   322→   322→    ctx.save(); ctx.setTransform(1, 0, 0, 1, 0, 0); band(-fw * 0.9, -fw * 2.2); ctx.clip();
   323→   323→    const sg = ctx.createLinearGradient(c[0] - dir[0] * fw * 0.9, c[1] - dir[1] * fw * 0.9, c[0] - dir[0] * fw * 2.2, c[1] - dir[1] * fw * 2.2); sg.addColorStop(0, 'rgba(30,10,2,0.6)'); sg.addColorStop(1, 'rgba(30,10,2,0)'); ctx.fillStyle = sg; ctx.fillRect(0, 0, W, H); ctx.restore();
   324→   324→    // the flap: mirror of the paper just behind the fold, seen from its back
   325→   325→    ctx.save(); ctx.setTransform(1, 0, 0, 1, 0, 0); band(0, -fw); ctx.clip();
   326→   326→    ctx.fillStyle = '#efe6d0'; ctx.fillRect(0, 0, W, H);
   327→   327→    ctx.globalCompositeOperation = 'multiply'; ctx.drawImage(paperTexture(W, H), 0, 0, W, H); ctx.globalCompositeOperation = 'source-over';
   328→   328→    const dx = dir[0], dy = dir[1], cd = c[0] * dx + c[1] * dy, M = [1 - 2 * dx * dx, -2 * dx * dy, -2 * dx * dy, 1 - 2 * dy * dy, 2 * cd * dx, 2 * cd * dy];
   329→   329→    ctx.globalAlpha = 0.28; ctx.filter = 'blur(0.8px)';
   330→   330→    drawPaperScene(ctx, t, cam, { noColour: true, pre: M, inkOnly: true });
   331→   331→    ctx.filter = 'none'; ctx.globalAlpha = 1; ctx.setTransform(1, 0, 0, 1, 0, 0);
   332→   332→    const fg = ctx.createLinearGradient(c[0], c[1], c[0] - dir[0] * fw, c[1] - dir[1] * fw);
   333→   333→    // the sheet bends like a cylinder: a dark crease at the fold, a bright roll, then it turns away from the light
   334→   334→    fg.addColorStop(0, 'rgba(70,52,28,0.55)'); fg.addColorStop(0.06, 'rgba(120,96,60,0.2)'); fg.addColorStop(0.22, 'rgba(255,252,242,0.55)'); fg.addColorStop(0.5, 'rgba(255,255,255,0.05)'); fg.addColorStop(0.85, 'rgba(120,96,60,0.25)'); fg.addColorStop(1, 'rgba(80,60,34,0.45)');
   335→   335→    ctx.fillStyle = fg; ctx.fillRect(0, 0, W, H);
   336→   336→    ctx.restore();
   337→   337→    ctx.save(); ctx.setTransform(1, 0, 0, 1, 0, 0); ctx.strokeStyle = 'rgba(80,60,34,0.5)'; ctx.lineWidth = 1.4; band(-fw, -fw - 0.01); ctx.stroke(); ctx.restore();
   338→   338→  }
